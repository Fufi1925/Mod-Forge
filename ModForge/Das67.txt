"""



 

 

ModForge – ZENTRALE EMBED KONFIGURATION (VOLLSTÄNDIG)



 

 

======================================================



 

 

 



 



Diese Datei enthält **JEDES** Embed, das der Bot jemals senden kann.



 

 

Inklusive aller DM-Embeds, Error-Embeds, Success-Embeds, Help-Embeds etc.



 

 

 



 



------------------------------------------------------------------



 

 

EINHEITLICHER AUFBAU – JEDES EMBED FOLGT DIESEM SCHEMA:



 

 

------------------------------------------------------------------



 

 

1. Author + Icon



 

 

   -&amp;amp;gt; Immer die Person, die die Aktion AUSGEFÜHRT hat



 

 

      (z.B. der Mod beim Ban, der Bot bei automatisierten Events).



 

 

      Name + Avatar dieser Person/des Bots werden als Author gesetzt.



 

 

 



 



2. Titel



 

 

 



 



3. Beschreibung



 

 

 



 



4. Felder im 2-Spalten-Layout:



 

 

       Feld 1 | Feld 2



 

 

       Feld 3 | Feld 4



 

 

   (wird automatisch über _add_fields() angeordnet)



 

 

 



 



5. Thumbnail (oben rechts)



 

 

   -&amp;amp;gt; Immer das Profilbild der Person, um die es im Embed geht



 

 

      (z.B. der gebannte User, der neue Member, ...).



 

 

 



 



6. Großes Bild



 

 

   -&amp;amp;gt; Nur wenn passend (z.B. Server-Banner bei Welcome-Embeds).



 

 

 



 



7. Footer + Icon



 

 

   -&amp;amp;gt; Immer "Powered by BotForge" + Bot-Icon.



 

 

 



 



8. Timestamp



 

 

   -&amp;amp;gt; Immer aktuell (UTC).



 

 

 



 



------------------------------------------------------------------



 

 

Verwendung:



 

 

    from bot.embed_config import get_embed



 

 

 



 



    embed = get_embed(



 

 

        "ban",



 

 

        user=user,          # betroffener User -&amp;amp;gt; Thumbnail



 

 

        mod=mod,            # ausführender Mod -&amp;amp;gt; Author



 

 

        reason=reason,



 

 

        guild=guild,



 

 

        bot=bot,            # für Footer-/Fallback-Icon



 

 

    )



 

 

    await channel.send(embed=embed)



 

 

"""

def get_embed(name: str, **kwargs) -&amp;amp;gt; discord.Embed:



 

 

    """Gibt ein zentral definiertes Embed zurück."""



 

 

    func = EMBEDS.get(name)



 

 

    if func:



 

 

        return func(**kwargs)



 

 

 



 



    # Fallback



 

 

    embed = discord.Embed(



 

 

        title="❌ Embed nicht gefunden",



 

 

        description=f"Das Embed `{name}` ist nicht definiert.",



 

 

        color=COLORS["danger"],



 

 

    )



 

 

    embed.set_footer(text=f"Powered by {BOT_NAME}", icon_url=BOT_ICON_URL)



 

 

    embed.timestamp = datetime.now(timezone.utc)



 

 

    return embed

def _avatar_url(entity: Any) -&amp;amp;gt; Optional[str]:



 

 

    """Liefert die Avatar-URL eines User/Member/ClientUser-Objekts."""



 

 

    if entity is None:



 

 

        return None



 

 

 



 



    avatar = getattr(entity, "display_avatar", None)



 

 

    if avatar is not None:



 

 

        return avatar.url



 

 

 



 



    avatar = getattr(entity, "avatar", None)



 

 

    if avatar is not None:



 

 

        return avatar.url



 

 

 



 



    default = getattr(entity, "default_avatar", None)



 

 

    if default is not None:



 

 

        return default.url



 

 

 



 



    return None



 

 

 



 



 



 

 

def _thumbnail_url(entity: Any) -&amp;amp;gt; Optional[str]:



 

 

    """



 

 

    Liefert die Thumbnail-URL für das "Worum geht es"-Objekt.



 

 

    Unterstützt User/Member (-&amp;amp;gt; Avatar) und Guild (-&amp;amp;gt; Server-Icon)



 

 

    als Fallback, falls kein konkreter User vorhanden ist.



 

 

    """



 

 

    if entity is None:



 

 

        return None



 

 

 



 



    if isinstance(entity, discord.Guild):



 

 

        icon = getattr(entity, "icon", None)



 

 

        return icon.url if icon else None



 

 

 



 



    return _avatar_url(entity)



 

 

 



 



 



 

 

def _bot_icon(bot: Any) -&amp;amp;gt; Optional[str]:



 

 

    """Icon für den Footer ('Powered by BotForge')."""



 

 

    if bot is not None:



 

 

        bot_user = getattr(bot, "user", bot)



 

 

        url = _avatar_url(bot_user)



 

 

        if url:



 

 

            return url



 

 

    return BOT_ICON_URL



 

 

 



 



 



 

 

def _bot_name(bot: Any) -&amp;amp;gt; str:



 

 

    if bot is not None:



 

 

        bot_user = getattr(bot, "user", bot)



 

 

        name = getattr(bot_user, "name", None)



 

 

        if name:



 

 

            return name



 

 

    return BOT_NAME



 

 

 



 



 



 

 

def _guild_banner(guild: Any) -&amp;amp;gt; Optional[str]:



 

 

    if guild is None:



 

 

        return None



 

 

    banner = getattr(guild, "banner", None)



 

 

    return banner.url if banner else None



 

 

 



 



 



 

 

def _fmt_dt(dt: Any, style: str = "F") -&amp;amp;gt; str:



 

 

    """Formatiert ein datetime-Objekt als dynamischen Discord-Timestamp."""



 

 

    if dt is None:



 

 

        return "Unbekannt"



 

 

    try:



 

 

        return f"&amp;amp;lt;t:{int(dt.timestamp())}:{style}&amp;amp;gt;"



 

 

    except (AttributeError, ValueError, OSError):



 

 

        return "Unbekannt"

# HELPER: Felder im 2-Spalten-Layout (Feld1 | Feld2 / Feld3 | Feld4)



 

 

# =========================================================



 

 

 



 



def _add_fields(embed: discord.Embed, fields: Optional[List[Dict[str, Any]]]) -&amp;amp;gt; None:



 

 

    """



 

 

    Fügt Felder zum Embed hinzu.



 

 

 



 



    - Felder ohne explizites "inline" werden automatisch im 2er-Raster



 

 

      angeordnet: Feld1 | Feld2 / Feld3 | Feld4 ...



 

 

    - Felder mit inline=False (z.B. lange Texte) belegen die volle Breite



 

 

      und schließen die aktuelle Zeile ab.



 

 

    """



 

 

    if not fields:



 

 

        return



 

 

 



 



    pending_inline = 0



 

 

    for f in fields:



 

 

        name = f.get("name") or "\u200b"



 

 

        value = f.get("value")



 

 

        if value is None or value == "":



 

 

            value = "—"



 

 

        inline = f.get("inline", True)



 

 

 



 



        if inline:



 

 

            embed.add_field(name=name, value=value, inline=True)



 

 

            pending_inline += 1



 

 

            if pending_inline == 2:



 

 

                pending_inline = 0



 

 

        else:



 

 

            # Offene Inline-Zeile mit unsichtbarem Spacer abschließen,



 

 

            # damit das Vollbreite-Feld in einer eigenen Zeile steht.



 

 

            if pending_inline == 1:



 

 

                embed.add_field(name="\u200b", value="\u200b", inline=True)



 

 

                pending_inline = 0



 

 

            embed.add_field(name=name, value=value, inline=False)



 

 

 



 



 



 

 

# =========================================================



 

 

# HELPER: Flexibles Embed bauen (zentrale Struktur)



 

 

# =========================================================



 

 

 



 



def _build_embed(



 

 

    title: str = "",



 

 

    description: str = "",



 

 

    color: int = COLORS["primary"],



 

 

    executor: Any = None,



 

 

    subject: Any = None,



 

 

    fields: Optional[List[Dict[str, Any]]] = None,



 

 

    image: Optional[str] = None,



 

 

    bot: Any = None,



 

 

    author_name: Optional[str] = None,



 

 

    author_icon: Optional[str] = None,



 

 

) -&amp;amp;gt; discord.Embed:



 

 

    """



 

 

    Erstellt ein Embed nach dem einheitlichen ModForge-Schema:



 

 

 



 



    Author (= executor)  -&amp;amp;gt;  Titel  -&amp;amp;gt;  Beschreibung



 

 

    -&amp;amp;gt;  Felder (2-spaltig)



 

 

    -&amp;amp;gt;  Thumbnail (= subject)



 

 

    -&amp;amp;gt;  Großes Bild (optional)



 

 

    -&amp;amp;gt;  Footer "Powered by BotForge"



 

 

    -&amp;amp;gt;  Timestamp



 

 

    """



 

 

    embed = discord.Embed(title=title, description=description, color=color)



 

 

 



 



    # ---------- Author = ausführende Person/System ----------



 

 

    if author_name is not None:



 

 

        embed.set_author(name=author_name, icon_url=author_icon)



 

 

    elif executor is not None:



 

 

        embed.set_author(name=str(executor), icon_url=_avatar_url(executor))



 

 

    elif bot is not None:



 

 

        embed.set_author(name=_bot_name(bot), icon_url=_bot_icon(bot))



 

 

 



 



    # ---------- Felder (2-spaltig) ----------



 

 

    _add_fields(embed, fields)



 

 

 



 



    # ---------- Thumbnail = betroffene Person/Server ----------



 

 

    thumb = _thumbnail_url(subject)



 

 

    if thumb:



 

 

        embed.set_thumbnail(url=thumb)



 

 

 



 



    # ---------- Großes Bild (nur wenn passend) ----------



 

 

    if image:



 

 

        embed.set_image(url=image)



 

 

 



 



    # ---------- Footer: immer "Powered by BotForge" ----------



 

 

    embed.set_footer(text=f"Powered by {BOT_NAME}", icon_url=_bot_icon(bot))



 

 

 



 



    # ---------- Timestamp ----------



 

 

    embed.timestamp = datetime.now(timezone.utc)



 

 

 



 



    return embed
