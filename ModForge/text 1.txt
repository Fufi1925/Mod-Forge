_login_attempts:Dict[str,deque]=defaultdict(deque)
_login_lock=threading.Lock()

def _ip()->str:
    return (request.headers.get("X-Forwarded-For",request.remote_addr or "?").split(",")[0].strip())

def _rate_blocked(ip:str)->bool:
    now=time.time()
    with _login_lock:
        dq=_login_attempts[ip]
        while dq and now-dq[0]>300: dq.popleft()
        return len(dq)>=8

def _record_fail(ip:str)->None:
    with _login_lock: _login_attempts[ip].append(time.time())

def _run_async(coro,timeout:float=8.0):
    if BOT_REF is None or not BOT_REF.loop or not BOT_REF.loop.is_running(): return None
    try: return asyncio.run_coroutine_threadsafe(coro,BOT_REF.loop).result(timeout)
    except Exception as ex: log.debug(f"_run_async: {ex}"); return None

def admin_required(fn:Callable)->Callable:
    @wraps(fn)
    def w(*a,**kw):
        if not session.get("admin"): return redirect(url_for("admin_login"))
        return fn(*a,**kw)
    return w

# Discord OAuth2 helpers
def _discord_api_call(path:str,token:str=None,method:str="GET",data:dict=None):
    url=f"{DISCORD_API}{path}"
    headers={"Content-Type":"application/x-www-form-urlencoded","Accept":"application/json"}
    if token: headers["Authorization"]=f"Bearer {token}"
    body=None
    if data:
        body=urllib.parse.urlencode(data).encode(); headers["Content-Type"]="application/x-www-form-urlencoded"
    req=urllib.request.Request(url,data=body,headers=headers,method=method)
    try:
        with urllib.request.urlopen(req,timeout=10) as resp: return json.loads(resp.read())
    except urllib.error.HTTPError as ex:
        log.error(f"Discord API {path}: {ex.code}")
        try: return json.loads(ex.read())
        except: return None
    except Exception as ex: log.error(f"Discord API {path}: {ex}"); return None

def _get_user_guilds_with_bot(user_guilds:list)->list:
    if not BOT_REF: return []
    bot_guild_ids={g.id for g in BOT_REF.guilds}
    result=[]
    for g in user_guilds:
        perms=g.get("permissions",0)
        try: perms=int(perms)
        except: perms=0
        if (perms & 0x20) and int(g["id"]) in bot_guild_ids:
            bg=BOT_REF.get_guild(int(g["id"]))
            result.append({"id":g["id"],"name":g["name"],"icon":g.get("icon",""),
                "member_count":bg.member_count if bg else 0,"has_bot":True})
    return result

def _bot_stats():
    gc=mc=up=lat=0
    try:
        if BOT_REF:
            gc=len(BOT_REF.guilds); mc=sum(g.member_count or 0 for g in BOT_REF.guilds)
            up=get_uptime(BOT_REF.start_time); lat=round(BOT_REF.latency*1000,1)
    except Exception: pass
    return gc,mc,up,lat

def _render(tmpl:str,**kw)->str:
    kw.setdefault("css",_CSS); kw.setdefault("topnav",_TOPNAV); kw.setdefault("footer",_FOOTER)
    kw.setdefault("scroll_js",_SCROLL_JS)
    return render_template_string(tmpl,**kw)

def _get_session_user():
    return session.get("discord_user")

def _user_can_manage_guild(guild_id:str)->bool:
    ug=session.get("user_guilds",[])
    for g in ug:
        if g.get("id")==guild_id:
            perms=int(g.get("permissions",0))
            return bool(perms & 0x20)
    return False

def _get_guild_config(guild_id:str)->dict:
    cfg=_run_async(BOT_REF.db.load_config(int(guild_id))) if BOT_REF else None
    return cfg or DEFAULT_CONFIG.copy()

@flask_app.route("/auth/discord")
def auth_discord():
    if not CLIENT_ID or not CLIENT_SECRET or not OAUTH2_REDIRECT:
        return redirect(url_for("discord_login_page"))
    state=secrets_mod.token_urlsafe(16); session["oauth_state"]=state
    params=urllib.parse.urlencode({"client_id":CLIENT_ID,"redirect_uri":OAUTH2_REDIRECT,
        "response_type":"code","scope":"identify guilds","state":state})
    return redirect(f"https://discord.com/oauth2/authorize?{params}")

@flask_app.route("/callback")
def oauth_callback():
    code=request.args.get("code"); state=request.args.get("state")
    if not code or state!=session.get("oauth_state"):
        return redirect(url_for("discord_login_page"))
    # Exchange code for token
    token_data=_discord_api_call("/oauth2/token",method="POST",data={
        "client_id":CLIENT_ID,"client_secret":CLIENT_SECRET,
        "grant_type":"authorization_code","code":code,"redirect_uri":OAUTH2_REDIRECT})
    if not token_data or "access_token" not in token_data:
        return _render(LOGIN_HTML,error="OAuth2 fehlgeschlagen. Versuche es erneut.",
                       info=None,admin_available=bool(ADMIN_PASSWORD))
    access_token=token_data["access_token"]
    # Get user & guilds
    user=_discord_api_call("/users/@me",token=access_token)
    guilds=_discord_api_call("/users/@me/guilds",token=access_token) or []
    if not user or "id" not in user:
        return _render(LOGIN_HTML,error="Nutzer-Daten konnten nicht abgerufen werden.",
                       info=None,admin_available=bool(ADMIN_PASSWORD))
    session.permanent=True
    session["discord_user"]=user; session["user_guilds"]=guilds
    session["access_token"]=access_token
    ACTIVITY.push("oauth_login",f"Discord-Login: {user.get('username','?')} ({user.get('id','')})")
    return redirect(url_for("user_dash_home"))

@flask_app.route("/dashboard/<guild_id>/api/save",methods=["POST"])
@user_required
def user_dash_save(guild_id:str):
    if not _user_can_manage_guild(guild_id):
        return jsonify({"error":"Insufficient permissions"}),403
    data=request.get_json(); module=data.get("module"); settings=data.get("settings")
    if not module or not isinstance(settings,dict):
        return jsonify({"error":"Invalid data"}),400
    if BOT_REF:
        try:
            cfg=_run_async(BOT_REF.db.load_config(int(guild_id))) or DEFAULT_CONFIG.copy()
            if module in cfg and isinstance(cfg[module],dict):
                cfg[module].update(settings)
            else:
                cfg[module]=settings
            _run_async(BOT_REF.db.set_config(int(guild_id),cfg))
            return jsonify({"ok":True})
        except Exception as ex:
            return jsonify({"error":str(ex)}),500
    return jsonify({"error":"Bot not ready"}),503

@flask_app.route("/admin/login",methods=["GET","POST"])
def admin_login():
    if not ADMIN_PASSWORD:
        return _render(ADMIN_LOGIN_HTML,error="ADMIN_PASSWORD nicht gesetzt.",css=_CSS)
    err=None; ip=_ip()
    if request.method=="POST":
        if _rate_blocked(ip): err="Zu viele Versuche. Bitte 5 Minuten warten."
        else:
            u=(request.form.get("username") or "").strip(); p=(request.form.get("password") or "")
            if secrets_mod.compare_digest(u,ADMIN_USERNAME) and secrets_mod.compare_digest(p,ADMIN_PASSWORD):
                session.permanent=True; session["admin"]=True
                ACTIVITY.push("admin_login",f"Admin-Login von {ip}")
                return redirect(url_for("admin_dashboard"))
            else: _record_fail(ip); err="Falsche Zugangsdaten."
    return _render(ADMIN_LOGIN_HTML,error=err,css=_CSS)
